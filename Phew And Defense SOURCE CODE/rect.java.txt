import java.awt.Rectangle;
import java.awt.Color;
import java.awt.event.KeyEvent;
import java.awt.Graphics;
import java.util.Timer;

public class rect extends Rectangle {

    Color color;

    // rocketSpeed sets how fast should the rockets move
    // Default: 7
    int rocketSpeed = 7;

    // bulletSpeed sets how fast should the bullet go
    // Default: 10
    int bulletSpeed = 10;

    rect(int x, int y, int width, int height, Color color) {
        this.x = x;
        this.y = y;
        this.width = width;
        this.height = height;
        this.color = color;
    }

    public void rocketUP() { // when rocketUP is called. it moves the rocket up
        y -= rocketSpeed;
    }

    public void rocketDOWN() { // when the rocketDOWN is called. it moves the rocket down
        y += rocketSpeed;
    }

    public void RABulletMOVE() { // when the RABulletMOVE is called. it makes the bullet move
        x += bulletSpeed;
    }

    public void RBBulletMOVE() { // when the RBBulletMOVE is called. it makes the bullet move
        x -= bulletSpeed;
    }

    public void draw(Graphics g) {
        g.setColor(this.color);
        g.fillRect(this.x, this.y, this.width, this.height);
    }
}
